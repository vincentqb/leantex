% A row of columns stands from the last line's baseline, each column's lines on the grid as a
% frame's, the columns flush with both margins; on a column grid each spans whole columns, the
% grid's gutter between them, each ending on the grid column nearest the row's width so far.
\int_new:N \l__lt_cols_int
\dim_new:N \l__lt_colw_dim
\dim_new:N \l__lt_gutter_dim
\bool_new:N \g__lt_colfirst_bool
\box_new:N \l__lt_row_box
\dim_new:N \g__lt_colsum_dim
\int_new:N \g__lt_coledge_int
\dim_new:N \l__lt_span_dim
\cs_new_protected:Npn \__lt_span:n #1
  {
    \int_compare:nNnTF { \l__lt_cols_int } > { 0 }
      {
        \dim_gadd:Nn \g__lt_colsum_dim {#1}
        \int_set:Nn \l_tmpa_int
          {
            \int_max:nn { \g__lt_coledge_int + 1 }
              {
                \int_min:nn { \l__lt_cols_int }
                  {
                    \fp_eval:n
                      {
                        floor ( \dim_to_fp:n { \g__lt_colsum_dim + \l__lt_gutter_dim }
                          / \dim_to_fp:n { \l__lt_colw_dim + \l__lt_gutter_dim } + 0.5 )
                      }
                  }
              }
          }
        \int_set:Nn \l_tmpb_int { \l_tmpa_int - \g__lt_coledge_int }
        \int_gset_eq:NN \g__lt_coledge_int \l_tmpa_int
        \dim_set:Nn \l__lt_span_dim
          { \l__lt_colw_dim * \l_tmpb_int + \l__lt_gutter_dim * ( \l_tmpb_int - 1 ) }
      }
      { \dim_set:Nn \l__lt_span_dim {#1} }
  }
