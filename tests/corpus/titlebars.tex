% A refused \maketitle redefinition still declares the built-in's styling
% (rule (b)'s remainder): the rules and skips around \@title land on the
% built-in's title block. Fixture-only numbers -- 3pt/2pt bars, 0.5in and
% 0.4in gaps, \Large -- differ from the NeurIPS lineage's 4pt/1pt and from
% every engine default, so the census can tell a declared value from a
% default. The definition order matches the venue's: \maketitle is redefined
% before the internals it names exist, which is why extraction reads the
% body at the preamble's end.
\documentclass{article}
\title{Bars Probe Title}
\author{An Invented Author}
\renewcommand{\maketitle}{\begingroup\thispagestyle{empty}\@maketitle\endgroup\let\maketitle\relax}
\newcommand{\@toptitlebar}{\hrule height 3pt \vskip 0.5in \vskip -\parskip}
\newcommand{\@bottomtitlebar}{\vskip 0.75in \vskip -\parskip \hrule height 2pt \vskip 0.1in}
\providecommand{\@maketitle}{}
\renewcommand{\@maketitle}{\vbox{\hsize\textwidth \vskip 0.1in \@toptitlebar \centering {\Large\bf \@title\par} \@bottomtitlebar \begin{tabular}[t]{c}\bf\rule{\z@}{24\p@}\@author\end{tabular} \vskip 0.3in \@minus 0.1in}}
\begin{document}
\maketitle

Body text follows the styled built-in title.
\end{document}
