\section{Plan the pattern}
\begin{frame}{Choose the pieces}
Use three paper shapes for a small mosaic.
\begin{itemize}
  \item A square provides the centre.
  \item Two triangles complete the border.
\end{itemize}
\end{frame}

\section{Build the pattern}
\begin{frame}{Arrange then attach}
Place the pieces on a blank card.

\pause
Attach each piece after checking the gaps.
\end{frame}

\setbeamertemplate{frame footer}{Workshop reminder}
\begin{frame}[standout]{Pause and compare}
Keep one clear gap.
\end{frame}

\setbeamertemplate{frame footer}{}
\begin{frame}{Pack the pieces}
Save the unused shapes for another pattern.
\end{frame}
